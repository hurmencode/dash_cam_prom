# ============================================================
#  dash_cam_prom — requirements.txt
# ============================================================
#
#  ВАЖНО: этот файл предполагает использование venv с флагом
#  --system-site-packages. На Linux/Jetson (Ubuntu 22.04/24.04,
#  Jetson Orin Nano) основные зависимости (OpenCV, NumPy, Pillow,
#  PyGObject, Aravis) должны быть установлены через apt, а НЕ
#  через pip. Иначе получишь конфликты и падения записи MJPG.
#
#  Правильная последовательность установки на Linux:
#
#    sudo apt update
#    sudo apt install -y \
#        python3-gi python3-gi-cairo gir1.2-aravis-0.10 \
#        libaravis-0.10-0 arv-tools \
#        python3-opencv python3-pil python3-numpy
#
#    python3 -m venv --system-site-packages .venv
#    source .venv/bin/activate
#    pip install -r requirements.txt
#
#  Для Windows/MSYS2 — см. README, раздел «Установка».
# ============================================================


# ------------------------------------------------------------
# Pillow — отображение кадров в Tkinter через ImageTk.
# На Linux идёт из apt как python3-pil. pip-версия совместима,
# но в venv с --system-site-packages устанавливать не нужно.
# ------------------------------------------------------------
Pillow>=10.0.0


# ------------------------------------------------------------
# NumPy — работа с кадрами Aravis (np.frombuffer, reshape).
# На Linux идёт из apt как python3-numpy. pip-версия совместима,
# но не требуется при --system-site-packages.
# ------------------------------------------------------------
numpy>=1.24.0


# ------------------------------------------------------------
# OpenCV — захват и запись видео (VideoWriter, MJPG, resize,
# cvtColor, putText). На Linux идёт из apt как python3-opencv.
#
# ВНИМАНИЕ: pip-пакет opencv-python НЕ РЕКОМЕНДУЕТСЯ на Jetson:
#   - он собран без Jetson-специфичных бэкендов,
#   - у него другой FFmpeg, из-за чего MJPG-запись ломается,
#   - cv2.getBuildInformation() покажет FFMPEG: NO, GSTREAMER: NO.
#
# Если системного python3-opencv нет — раскомментируй строку ниже
# ТОЛЬКО для обычного x86_64 Ubuntu, не для Jetson.
# ------------------------------------------------------------
# opencv-python>=4.8.0


# ------------------------------------------------------------
# PyGObject (gi) — не ставится через pip без головной боли:
# нужны libgirepository1.0-dev, libcairo2-dev, компилятор.
# На Linux ставится через apt как python3-gi.
# В requirements.txt отсутствует намеренно.
# ------------------------------------------------------------

#   MSYS2:   pacman -S mingw-w64-x86_64-aravis
# Сборка из исходников — см. README, раздел «Сборка Aravis».

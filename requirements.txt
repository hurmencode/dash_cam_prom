# ============================================================
#  dash_cam_prom — зависимости
# ============================================================
#
#  На Jetson (JetPack 6 = Ubuntu 22.04, Python 3.10) всё, кроме Pillow,
#  ставится через apt, а НЕ через pip:
#
#    sudo apt update
#    sudo apt install -y \
#        python3-gi python3-gi-cairo gir1.2-aravis-0.8 arv-tools \
#        python3-opencv python3-numpy python3-pil python3-tk \
#        v4l-utils ffmpeg
#
#    python3 -m venv --system-site-packages .venv
#    source .venv/bin/activate
#    pip install -r requirements.txt
#
#  Почему не pip-версии:
#   * pip-пакет opencv-python на aarch64 собран без Jetson-специфики и
#     может конфликтовать с системным python3-opencv из JetPack;
#   * PyGObject (gi) и Aravis через pip не ставятся вообще.
#
#  ffmpeg нужен для режимов записи «как есть» (MJPG от USB-камеры)
#  и FFV1 (lossless). Для обычного MJPG через OpenCV не нужен.
# ============================================================

# Минимальные версии; при --system-site-packages уже установленные
# apt-пакеты удовлетворяют этим требованиям и pip их не трогает.

numpy>=1.21.0
